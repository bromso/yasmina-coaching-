one brand constant, a guard test, and a publish workflow that builds images for the owner's domain.

app removals rehearsed on a throwaway local branch.

**Tech Stack:** Bun 1.4.2 workspaces, Next 16.2, `bun test`, Docker (`docker/next.Dockerfile`,
`docker/bun-service.Dockerfile`), GitHub Actions.

- [ ] **Step 1: Write the guard and its unit tests**

`scripts/brand.ts`:

```ts
import { readFileSync } from "node:fs"

/** The literal brand name; only `packages/sites/src/brand.ts` may spell it out in first-party src. */
export const LITERAL = "Boilerplate"

/** Files that may contain the literal, each with its reason. */
export const ALLOW = ["packages/sites/src/brand.ts"]

const isTest = (path: string) => /\.test\.(ts|tsx)$/.test(path)

/** `<path>:<line>` for every line naming the brand outside the allow-list and test files. */
export function findBrandLiterals(files: { path: string; text: string }[], allow = ALLOW) {
  const hits: string[] = []
  for (const { path, text } of files) {
    if (allow.includes(path) || isTest(path)) continue
    text.split("\n").forEach((line, i) => {
      if (line.includes(LITERAL)) hits.push(`${path}:${i + 1}`)
    })
  }
  return hits
}

/** Tracked first-party source: every app's and package's `src/`, TypeScript only. */
export function firstPartySources(): { path: string; text: string }[] {
  const out = Bun.spawnSync(["git", "ls-files", "apps/*/src/**", "packages/*/src/**"]).stdout
  return new TextDecoder()
    .decode(out)
    .split("\n")
    .filter((p) => /\.(ts|tsx)$/.test(p))
    .map((path) => ({ path, text: readFileSync(path, "utf8") }))
}
```

`scripts/brand.test.ts`:

```ts
import { expect, test } from "bun:test"
import { findBrandLiterals, firstPartySources } from "./brand"

test("finds the literal in JSX text and template strings", () => {
  const hits = findBrandLiterals([
    { path: "apps/x/src/a.tsx", text: "<p>\n  © 2026 Boilerplate\n</p>" },
    { path: "apps/x/src/b.ts", text: "const t = `%s | Boilerplate`" },
  ])
  expect(hits).toEqual(["apps/x/src/a.tsx:2", "apps/x/src/b.ts:1"])
})

test("skips test files and the allow-list", () => {
  const hits = findBrandLiterals([
    { path: "packages/sites/src/brand.ts", text: 'export const BRAND_NAME = "Boilerplate"' },
    { path: "apps/x/src/a.test.tsx", text: "Boilerplate" },
  ])
  expect(hits).toEqual([])
})

test("first-party src names the brand only through BRAND_NAME", () => {
  expect(findBrandLiterals(firstPartySources())).toEqual([])
})
```
